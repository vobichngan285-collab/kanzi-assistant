cmake_minimum_required(VERSION 3.22)
project(bluestar_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8.6-a+dotprod+i8mm+fp16+bf16" CACHE STRING "" FORCE)
set(GGML_CPU_KLEIDIAI ON CACHE BOOL "" FORCE)
set(GGML_OPENMP ON CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_OPENCL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_HTML OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_UI OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(ENABLE_QNN OFF CACHE BOOL "" FORCE)
if(NOT EXISTS "${ENGINE_DIR}/tools/mtmd/mtmd.cpp")
  message(FATAL_ERROR "Pinned engine missing")
endif()
add_subdirectory("${ENGINE_DIR}" engine EXCLUDE_FROM_ALL)
set(LLAMA_INSTALL_VERSION "0.0.0")
add_subdirectory("${ENGINE_DIR}/tools/mtmd" mtmd EXCLUDE_FROM_ALL)
add_library(bluestar_native SHARED native.cpp)
target_include_directories(bluestar_native PRIVATE "${ENGINE_DIR}/common" "${ENGINE_DIR}/vendor")
target_link_libraries(bluestar_native PRIVATE llama llama-common mtmd android log)
